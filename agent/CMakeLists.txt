cmake_minimum_required(VERSION 3.20)
project(HyperFamilyStoreAgent LANGUAGES CXX RC)

if(NOT WIN32)
  message(FATAL_ERROR "The agent targets Windows. Use a Windows build machine or a Windows cross-toolchain.")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Build the agent for Windows x64.")
endif()
if(NOT DEFINED AGENT_VERSION OR NOT AGENT_VERSION MATCHES "^[0-9]+\\.[0-9]+\\.[0-9]+(-[A-Za-z0-9.-]+)?$")
  message(FATAL_ERROR "Pass -DAGENT_VERSION=<package.json version> (or use npm run build:agent).")
endif()
string(REGEX MATCH "^([0-9]+)\\.([0-9]+)\\.([0-9]+)" VERSION_PARTS "${AGENT_VERSION}")
set(AGENT_FILE_VERSION "${CMAKE_MATCH_1},${CMAKE_MATCH_2},${CMAKE_MATCH_3},0")
# Stable builds must not carry the Windows prerelease resource flag.
set(AGENT_FILE_FLAGS "0")
if(AGENT_VERSION MATCHES "-")
  set(AGENT_FILE_FLAGS "VS_FF_PRERELEASE")
endif()
configure_file(agent_version.hpp.in generated/agent_version.hpp @ONLY)
configure_file(agent.rc.in generated/agent.rc @ONLY)

add_executable(HyperFamilyStoreAgent WIN32 main.cpp "${CMAKE_CURRENT_BINARY_DIR}/generated/agent.rc")
target_include_directories(HyperFamilyStoreAgent PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_features(HyperFamilyStoreAgent PRIVATE cxx_std_17)
target_compile_definitions(HyperFamilyStoreAgent PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00)
target_link_libraries(HyperFamilyStoreAgent PRIVATE advapi32 ole32 shell32)
set_target_properties(HyperFamilyStoreAgent PROPERTIES
  RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/build"
  RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_SOURCE_DIR}/build")

if(MSVC)
  string(REPLACE "/O2" "/O1" CMAKE_CXX_FLAGS_RELEASE "${CMAKE_CXX_FLAGS_RELEASE}")
endif()

include(CTest)
add_executable(AgentJsonTests tests/json.test.cpp)
target_compile_features(AgentJsonTests PRIVATE cxx_std_17)
add_test(NAME AgentJsonTests COMMAND AgentJsonTests)

foreach(target HyperFamilyStoreAgent AgentJsonTests)
  if(MSVC)
    # Static CRT: no .NET, VC++ Redistributable or extra application DLLs.
    set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(${target} PRIVATE /W4 /WX /EHsc /permissive- /utf-8 /O1 /guard:cf)
    target_link_options(${target} PRIVATE /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA /guard:cf /OPT:REF /OPT:ICF /INCREMENTAL:NO)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -Os)
    target_link_options(${target} PRIVATE -static -static-libgcc -static-libstdc++ -s)
  endif()
endforeach()
if(MINGW)
  target_link_options(HyperFamilyStoreAgent PRIVATE -municode)
endif()
